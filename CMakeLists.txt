cmake_minimum_required(VERSION 3.24)
project(MouseOptions VERSION 0.2.0 LANGUAGES CXX)
add_executable(MouseOptions WIN32 src/main.cpp src/settings.cpp src/integration.cpp src/meteors.cpp src/meteor_settings.cpp)
target_compile_features(MouseOptions PRIVATE cxx_std_20)
target_compile_definitions(MouseOptions PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(MouseOptions PRIVATE /W4 /permissive- /utf-8 /O1 /Gy)
set_property(TARGET MouseOptions PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(MouseOptions PRIVATE user32 gdi32 shell32 comctl32 comdlg32 advapi32 shcore)
target_link_options(MouseOptions PRIVATE /OPT:REF /OPT:ICF /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/app.manifest")
install(TARGETS MouseOptions RUNTIME DESTINATION .)
install(FILES README.md DESTINATION .)
install(FILES docs/manual-checks.md DESTINATION . RENAME MANUAL-CHECKS.md)
